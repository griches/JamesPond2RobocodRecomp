cmake_minimum_required(VERSION 3.20)
project(JamesPond2Recomp VERSION 1.0.3 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)   # also clang-cl: plain C library calls without Microsoft's "unsafe" warnings
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# The player's disk image. The recompiled build is generated from it; without
# it only the interpreter build and the tools are made.
set(JP2_ADF "" CACHE FILEPATH "James Pond 2 ADF (Skid Row / Kixx release) to recompile")
# Development: each play session records the code that had to be interpreted
# into coverage/, for the next recompile. Off for release builds.
option(JP2_RECORD_PLAY "Record each play session's code coverage into coverage/" ON)

# The PlayStation 5's build (ps5/cmake/ps5-game.cmake) brings its own SDL3.
if(NOT TARGET SDL3::SDL3)
  find_package(SDL3 REQUIRED CONFIG)
endif()

# Musashi 68000 interpreter (MIT). m68kops.c/h are generated by m68kmake.
set(MUSASHI ${CMAKE_SOURCE_DIR}/vendor/musashi)
add_library(musashi STATIC
  ${MUSASHI}/m68kcpu.c
  ${MUSASHI}/m68kdasm.c
  ${MUSASHI}/gen/m68kops.c
  ${MUSASHI}/softfloat/softfloat.c)
target_include_directories(musashi PUBLIC ${MUSASHI} ${MUSASHI}/gen ${CMAKE_SOURCE_DIR}/runtime)
target_compile_definitions(musashi PUBLIC MUSASHI_CNF="musashi_conf.h")
target_compile_options(musashi PRIVATE -w)

# The menu's fonts (Inter, SIL OFL; Material Symbols, Apache 2.0; res/fonts), built into the program.
set(FONT_SRC)
foreach(font inter_regular:Inter-Regular inter_semibold:Inter-SemiBold icons:MaterialSymbolsRounded-subset)
  string(REPLACE ":" ";" parts ${font})
  list(GET parts 0 sym)
  list(GET parts 1 file)
  set(out ${CMAKE_BINARY_DIR}/font_${sym}.cpp)
  add_custom_command(
    OUTPUT ${out}
    COMMAND ${CMAKE_COMMAND} -DIN=${CMAKE_SOURCE_DIR}/res/fonts/${file}.ttf -DOUT=${out} -DNAME=font_${sym}
            -P ${CMAKE_SOURCE_DIR}/tools/embed.cmake
    DEPENDS ${CMAKE_SOURCE_DIR}/res/fonts/${file}.ttf ${CMAKE_SOURCE_DIR}/tools/embed.cmake)
  list(APPEND FONT_SRC ${out})
endforeach()

set(JP2_RUNTIME
  ${FONT_SRC}
  runtime/blitter.cpp
  runtime/cia.cpp
  runtime/custom.cpp
  runtime/disk.cpp
  vendor/puff/puff.c
  runtime/display.cpp
  runtime/game.cpp
  runtime/glyphs.cpp
  runtime/main.cpp
  runtime/memory.cpp
  runtime/menu.cpp
  runtime/musashi_bus.cpp
  runtime/overlay.cpp
  runtime/paula.cpp
  runtime/png.cpp
  runtime/record.cpp
  runtime/settings.cpp
  runtime/setup.cpp
  runtime/state.cpp
  runtime/trace.cpp
  runtime/ui.cpp
  runtime/wide.cpp)

# The PlayStation 5 and Android build the game alone, from code another build
# recompiled (JP2_GENERATED_DIR: the Mac's build/mac/generated).
if(JP2_PS5 OR ANDROID)
  set(GEN ${JP2_GENERATED_DIR})
  file(GLOB GEN_CODE ${GEN}/code_*.cpp)
  add_library(jp2-code OBJECT ${GEN_CODE} ${GEN}/tables.cpp)
  target_include_directories(jp2-code PRIVATE runtime)
  target_compile_options(jp2-code PRIVATE -Wno-unused-label -Wno-unused-variable -Wno-parentheses-equality)
  set(JP2_GAME ${JP2_RUNTIME} runtime/cpu_recomp.cpp runtime/recomp_exc.cpp $<TARGET_OBJECTS:jp2-code>)
  if(ANDROID)   # SDL's Java side loads the game as libmain.so
    add_library(jp2 SHARED ${JP2_GAME})
    set_target_properties(jp2 PROPERTIES OUTPUT_NAME main)
  else()
    add_executable(jp2 ${JP2_GAME})
  endif()
  target_include_directories(jp2 PRIVATE runtime vendor/stb vendor/puff)
  target_link_libraries(jp2 PRIVATE musashi SDL3::SDL3)
  return()
endif()

# Interpreter build: the reference, and the way to record code coverage.
add_executable(jp2-interp ${JP2_RUNTIME} runtime/cpu_musashi.cpp)
target_include_directories(jp2-interp PRIVATE runtime vendor/stb vendor/puff)
target_link_libraries(jp2-interp PRIVATE musashi SDL3::SDL3)

# The recompiler.
add_executable(jp2-recomp tools/recomp/recomp.cpp tools/recomp/translate.cpp runtime/disk.cpp vendor/puff/puff.c)
target_include_directories(jp2-recomp PRIVATE runtime tools/recomp vendor/puff)
target_link_libraries(jp2-recomp PRIVATE musashi)

if(JP2_ADF)
  set(GEN ${CMAKE_BINARY_DIR}/generated)
  set(GEN_CODE)
  foreach(i RANGE 0 31)
    list(APPEND GEN_CODE ${GEN}/code_${i}.cpp)
  endforeach()
  # Coverage recorded with `jp2-interp --record dir` (coverage.bin) helps the
  # recompiler find code reached only through jump tables and pointers.
  file(GLOB_RECURSE COVERAGE_FILES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/coverage/*.bin)
  set(COVERAGE_ARGS)
  foreach(c ${COVERAGE_FILES})
    list(APPEND COVERAGE_ARGS --coverage ${c})
  endforeach()
  add_custom_command(
    OUTPUT ${GEN_CODE} ${GEN}/tables.cpp ${GEN}/tests.cpp
    COMMAND ${CMAKE_COMMAND} -E make_directory ${GEN}
    COMMAND jp2-recomp --adf ${JP2_ADF} --out ${GEN} --test-out ${GEN}/tests.cpp
            --break 16E00 --break 16D66 --break 16D9A --break 13DC6 --break 152C2 --break 4BA24 --break 1490C --break 11B18 ${COVERAGE_ARGS}
    DEPENDS jp2-recomp ${JP2_ADF} ${COVERAGE_FILES}
    COMMENT "Recompiling the 68000 code from ${JP2_ADF}")

  add_library(jp2-code OBJECT ${GEN_CODE} ${GEN}/tables.cpp)
  target_include_directories(jp2-code PRIVATE runtime)
  target_compile_options(jp2-code PRIVATE -Wno-unused-label -Wno-unused-variable -Wno-parentheses-equality)

  add_executable(jp2 ${JP2_RUNTIME} runtime/cpu_recomp.cpp runtime/recomp_exc.cpp $<TARGET_OBJECTS:jp2-code>)
  target_include_directories(jp2 PRIVATE runtime vendor/stb vendor/puff)
  target_link_libraries(jp2 PRIVATE musashi SDL3::SDL3)
  if(JP2_RECORD_PLAY)
    target_compile_definitions(jp2 PRIVATE JP2_COVERAGE_DIR="${CMAKE_SOURCE_DIR}/coverage")
  endif()
  set_target_properties(jp2 PROPERTIES WIN32_EXECUTABLE ON)   # Windows: a window, no console

  add_executable(jp2-difftest tools/recomp/difftest.cpp runtime/disk.cpp vendor/puff/puff.c runtime/recomp_exc.cpp
                 ${GEN}/tests.cpp $<TARGET_OBJECTS:jp2-code>)
  target_include_directories(jp2-difftest PRIVATE runtime vendor/puff)
  target_compile_options(jp2-difftest PRIVATE -Wno-unused-label -Wno-unused-variable -Wno-parentheses-equality)
  target_link_libraries(jp2-difftest PRIVATE musashi)
endif()

# Double-clickable app (macOS). It asks for the disk on first run.
if(APPLE AND JP2_ADF)
  set(JP2_APP "${CMAKE_BINARY_DIR}/James Pond 2.app")
  add_custom_command(
    OUTPUT "${JP2_APP}/Contents/MacOS/jp2"
    COMMAND ${CMAKE_SOURCE_DIR}/tools/make_mac_app.sh $<TARGET_FILE:jp2> "${JP2_APP}" ${PROJECT_VERSION}
    DEPENDS jp2 ${CMAKE_SOURCE_DIR}/tools/make_mac_app.sh
    COMMENT "Making James Pond 2.app")
  add_custom_target(jp2-app ALL DEPENDS "${JP2_APP}/Contents/MacOS/jp2")
endif()
